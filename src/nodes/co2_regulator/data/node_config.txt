NODE_TANK_ID=0
NODE_NAME=UnmappedCO2
FIRMWARE_VERSION=1

# ============================================================================
# ESP-NOW SETTINGS

ESPNOW_CHANNEL=11

# ============================================================================
# HARDWARE PIN CONFIGURATION (ESP8266 D1 Mini)
# ============================================================================
# RELAY_PIN: GPIO connected to CO2 solenoid relay module
#   D0/GPIO16 = default. HIGH = CO2 ON, LOW = CO2 OFF
RELAY_PIN=16

# PH_SENSOR_PIN: Analog input for DFRobot pH sensor board
#   A0 (GPIO17) = ONLY analog input on ESP8266. Do not change unless using
#   an external ADC or I2C-based pH sensor board.
PH_SENSOR_PIN=17

# STATUS_LED_PIN: Status indicator LED
#   D5/GPIO14 = used instead of default D7/GPIO13 (which is the relay pin)
STATUS_LED_PIN=14

# ============================================================================
# pH READING CONFIGURATION
# ============================================================================
# PH_READ_INTERVAL_MS: How often to read and report pH to hub (milliseconds)
#   Default: 300000 (5 minutes). Hub analyzes trend over time.
#   Minimum recommended: 60000 (1 min). Maximum: 3600000 (1 hour)
PH_READ_INTERVAL_MS=300000

# PH_SAMPLE_COUNT: Number of ADC samples per pH reading (for noise filtering)
#   Samples are sorted and outliers discarded. Range: 3-50
PH_SAMPLE_COUNT=10

# PH_DISCARD_COUNT: Number of samples to discard from EACH end (high+low outliers)
#   Must be < PH_SAMPLE_COUNT/2. Set 0 to disable outlier removal.
PH_DISCARD_COUNT=2

# ============================================================================
# pH CALIBRATION
# ============================================================================
# pH Regulator Board Calibration
# Formula: pH = PH_CALIB_SLOPE * voltage + PH_CALIB_OFFSET
#
# Board manual specs (nominal, before probe calibration):
#   pH 4.0 = 1.5V output
#   pH 7.0 = 2.0V output
#   pH 9.0 = 2.5V output
#
# Two-point calibration using pH4 and pH7:
#   slope = (7.0 - 4.0) / (2.0 - 1.5) = 6.0
#   offset = 4.0 - 6.0 * 1.5 = -5.0
#
# To recalibrate with buffer solutions:
#   1. Place probe in pH 7.0 buffer, note voltage from serial output
#   2. Place probe in pH 4.0 buffer, note voltage from serial output
#   3. Calculate: slope = (pH2-pH1)/(V2-V1), offset = pH1 - slope*V1
#   4. Update values below and reboot
PH_CALIB_OFFSET=-5.0
PH_CALIB_SLOPE=6.0

# PH_CALIB_VOLT_TRIM: Fine-tune voltage offset (volts) applied BEFORE slope/offset.
#   Formula with trim: pH = PH_CALIB_SLOPE * (V_raw + PH_CALIB_VOLT_TRIM) + PH_CALIB_OFFSET
#   Use hub command 41 to calibrate automatically: place probe in buffer solution,
#   send {cmd:41, actual_ph:<buffer_ph>} and the node will compute and save the trim.
#   Manual calculation: trim = (actual_pH - PH_CALIB_OFFSET) / PH_CALIB_SLOPE - V_measured
#   Example: reading pH 7.47 when actual is 7.00, V_measured = 2.079V
#     trim = (7.00 - (-5.0)) / 6.0 - 2.079 = 2.000 - 2.079 = -0.0783
PH_CALIB_VOLT_TRIM=-0.0783

# ADC reference voltage and resolution
# D1 Mini has voltage divider on A0: 0-3.3V input range
ADC_VOLTAGE_REF=3.3
ADC_MAX_VALUE=1024

# ============================================================================
# pH PIPELINE CONFIGURATION (continuous median-filtered reading)
# ============================================================================
# The pH pipeline reads A0 continuously and computes medians:
#   1. Read A0 every PH_ADC_READ_INTERVAL_MS (200ms)
#   2. After PH_SAMPLES_PER_MEDIAN readings (20), compute short-term median
#   3. Accumulate medians over PH_TREND_WINDOW_SEC (300s)
#   4. Compute trend median for hub reporting
#
# Timing: 20 samples x 200ms = 4s per short-term median
#         300s / 4s = 75 medians per trend window

# How often to read A0 (milliseconds). Range: 50-10000
PH_ADC_READ_INTERVAL_MS=200

# Number of readings per short-term median batch. Range: 3-20
PH_SAMPLES_PER_MEDIAN=20

# Trend window duration in seconds. Range: 10-3600
PH_TREND_WINDOW_SEC=300

# ============================================================================
# LOCAL SAFETY LIMITS (Emergency protection independent of hub)
# ============================================================================
# PH_SAFETY_LOW: If pH drops below this, CO2 is FORCED OFF immediately
#   This is a local emergency cutoff. Hub handles trend-based control.
#   Set lower than your normal operating range to avoid false triggers.
PH_SAFETY_LOW=5.0

# PH_SAFETY_HIGH: High pH warning threshold (informational only, logged)
PH_SAFETY_HIGH=8.5

# ============================================================================
# DEBUG SETTINGS
# ============================================================================
DEBUG_SERIAL=true
DEBUG_ESPNOW=true
DEBUG_HARDWARE=false

# ============================================================================
# TIMING SETTINGS

ANNOUNCE_INTERVAL_MS=5000
HEARTBEAT_INTERVAL_MS=30000
CONNECTION_TIMEOUT_MS=90000

# Hub heartbeat timeout (ms) - if node does not receive hub heartbeat for this
# long, enter fail-safe mode (CO2 OFF). Default: 600000 ms (10 minutes)
HUB_HEARTBEAT_TIMEOUT_MS=600000

# ============================================================================
# CO2 REGULATOR HARDWARE NOTES
# ============================================================================
# Platform: ESP8266 D1 Mini (Wemos)
#
# pH Sensor: DFRobot Gravity Analog pH Sensor V2
#   - BNC connector: pH probe -> sensor board
#   - Sensor board signal -> A0 (analog output 0-3V)
#   - VCC: 3.3V or 5V, GND
#
# CO2 Relay: Standard relay module
#   D0/GPIO16 -> relay IN (HIGH = energized = CO2 flows)
#   - Relay controls CO2 solenoid valve
#
# Safety: CO2 is ALWAYS OFF at boot and on any error/disconnect
#
# Pin Map (D1 Mini):
#   A0  (GPIO17) = pH sensor analog input
#   D0  (GPIO16) = CO2 relay output
#   D5  (GPIO14) = Status LED
#   D1  (GPIO5)  = Available (I2C SCL)
#   D2  (GPIO4)  = Available (I2C SDA)
#
# Command Types from Hub:
#   0  = Emergency stop (CO2 OFF)
#   1  = CO2 ON (relay HIGH)
#   2  = CO2 OFF (relay LOW)
#   40 = Status request (read pH now)
#   41 = Calibrate: data[1..4] = known actual pH (float)
#        Node computes voltage trim and saves PH_CALIB_VOLT_TRIM
#   0xFF = Force fail-safe (test)
